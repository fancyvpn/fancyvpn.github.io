// Fancy Stats — приём событий от приложений и страница статистики.
// Привязки: DB (D1 fancy-stats) и секрет STATS_PASSWORD (пароль страницы).
// APP_TOKEN вшит в приложение, поэтому не секрет — храним прямо здесь.
const APP_TOKEN = "PosWdWkApno5R2NX8FIixb7lNNODT5Ty";

const DAY = 86400000;
const EVENTS = {
  launch: "launches",
  banner_ok: "banner_shown",
  banner_fail: "banner_fail",
  reward_ok: "reward_shown",
  reward_fail: "reward_fail",
};
const APPS = { vpn: "Fancy VPN" };

export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    if (req.method === "POST" && url.pathname === "/e") return collect(req, env);
    if (req.method === "GET" && url.pathname === "/") return stats(req, env, url);
    return new Response("not found", { status: 404 });
  },
};

async function collect(req, env) {
  if (req.headers.get("X-App-Token") !== APP_TOKEN) return new Response("forbidden", { status: 403 });
  let b;
  try { b = await req.json(); } catch { return new Response("bad json", { status: 400 }); }
  const app = String(b.app || "");
  const id = String(b.id || "");
  const col = EVENTS[b.ev];
  if (!APPS[app] || !/^[0-9a-f]{64}$/.test(id) || !col) return new Response("bad", { status: 400 });
  const now = Date.now();
  const version = String(b.v || "").slice(0, 20);
  const trial = Math.max(0, Math.min(Number(b.trial_until) || 0, now + 365 * DAY));
  const premium = b.premium ? 1 : 0;
  await env.DB.prepare(
    `INSERT INTO devices (app,id,version,first_seen,last_seen,trial_until,premium,${col})
     VALUES (?1,?2,?3,?4,?4,?5,?6,1)
     ON CONFLICT(app,id) DO UPDATE SET version=?3,last_seen=?4,trial_until=?5,premium=?6,${col}=${col}+1`
  ).bind(app, id, version, now, trial, premium).run();
  return new Response("ok");
}

function authorized(req, env) {
  const h = req.headers.get("Authorization") || "";
  if (!h.startsWith("Basic ") || !env.STATS_PASSWORD) return false;
  let decoded = "";
  try { decoded = atob(h.slice(6)); } catch { return false; }
  const pass = decoded.slice(decoded.indexOf(":") + 1);
  return pass === env.STATS_PASSWORD;
}

const esc = (s) => String(s ?? "").replace(/[&<>"]/g, (c) => ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;" }[c]));

function ago(ms, now) {
  const m = Math.floor((now - ms) / 60000);
  if (m < 1) return "только что";
  if (m < 60) return `${m} мин назад`;
  const h = Math.floor(m / 60);
  if (h < 24) return `${h} ч назад`;
  return `${Math.floor(h / 24)} дн назад`;
}

async function stats(req, env, url) {
  if (!authorized(req, env)) {
    return new Response("Нужен пароль", {
      status: 401,
      headers: { "WWW-Authenticate": 'Basic realm="Fancy Stats", charset="UTF-8"' },
    });
  }
  const app = APPS[url.searchParams.get("app")] ? url.searchParams.get("app") : "vpn";
  const { results } = await env.DB.prepare(
    "SELECT * FROM devices WHERE app=?1 ORDER BY last_seen DESC LIMIT 5000"
  ).bind(app).all();
  const now = Date.now();
  const sum = (k) => results.reduce((a, r) => a + (r[k] || 0), 0);
  const active1 = results.filter((r) => now - r.last_seen < DAY).length;
  const active7 = results.filter((r) => now - r.last_seen < 7 * DAY).length;

  const rows = results.map((r) => {
    const left = r.trial_until > now ? Math.ceil((r.trial_until - now) / DAY) : 0;
    const trial = r.premium ? "—" : left > 0 ? left : "Смотрит рекламу";
    return `<tr>
      <td class="mono">${esc(r.id.slice(0, 6))}</td>
      <td>${esc(r.version)}</td>
      <td title="${new Date(r.last_seen).toISOString()}">${ago(r.last_seen, now)}</td>
      <td class="${left > 0 ? "" : "ad"}">${trial}</td>
      <td>${r.premium ? "★ да" : ""}</td>
      <td class="n">${r.launches}</td>
      <td class="n">${r.banner_shown}</td>
      <td class="n ${r.banner_fail ? "bad" : ""}">${r.banner_fail}</td>
      <td class="n">${r.reward_shown}</td>
      <td class="n ${r.reward_fail ? "bad" : ""}">${r.reward_fail}</td>
    </tr>`;
  }).join("");

  const html = `<!doctype html><html lang="ru"><head><meta charset="utf-8">
<meta name="viewport" content="width=device-width,initial-scale=1">
<title>Статистика — ${esc(APPS[app])}</title>
<style>
:root{--bg:#f6f6f8;--card:#fff;--text:#1a1a1f;--muted:#6b6b76;--line:#e4e4ea;--bad:#c62828;--ad:#7a4fd6}
@media (prefers-color-scheme:dark){:root{--bg:#121216;--card:#1c1c22;--text:#ececf1;--muted:#9a9aa6;--line:#2c2c35;--bad:#ff6b6b;--ad:#b38cff}}
body{margin:0;background:var(--bg);color:var(--text);font:14px/1.4 system-ui,-apple-system,Segoe UI,Roboto,sans-serif}
main{max-width:1200px;margin:0 auto;padding:16px}
h1{font-size:20px;margin:4px 0 12px}
.kpi{display:flex;flex-wrap:wrap;gap:8px;margin-bottom:12px}
.kpi div{background:var(--card);border:1px solid var(--line);border-radius:10px;padding:8px 12px}
.kpi b{display:block;font-size:20px}.kpi span{color:var(--muted);font-size:12px}
.wrap{overflow-x:auto;background:var(--card);border:1px solid var(--line);border-radius:10px}
table{border-collapse:collapse;width:100%;min-width:880px}
th,td{padding:7px 10px;border-bottom:1px solid var(--line);text-align:left;white-space:nowrap}
th{font-size:12px;color:var(--muted);font-weight:600;position:sticky;top:0;background:var(--card)}
td.n,th.n{text-align:right;font-variant-numeric:tabular-nums}
.mono{font-family:ui-monospace,Menlo,Consolas,monospace}
.bad{color:var(--bad);font-weight:600}.ad{color:var(--ad)}
.muted{color:var(--muted);font-size:12px;margin-top:8px}
</style></head><body><main>
<h1>${esc(APPS[app])} — статистика по телефонам</h1>
<div class="kpi">
 <div><b>${results.length}</b><span>телефонов всего</span></div>
 <div><b>${active1}</b><span>активны за сутки</span></div>
 <div><b>${active7}</b><span>активны за 7 дней</span></div>
 <div><b>${sum("launches")}</b><span>запусков</span></div>
 <div><b>${sum("banner_shown")}</b><span>баннеров показано</span></div>
 <div><b>${sum("reward_shown")}</b><span>роликов показано</span></div>
 <div><b>${sum("banner_fail") + sum("reward_fail")}</b><span>ошибок рекламы</span></div>
</div>
<div class="wrap"><table>
<thead><tr><th>Код телефона</th><th>Версия</th><th>Последняя активность</th><th>Дней без рекламы</th><th>Premium</th>
<th class="n">Запуски</th><th class="n">Баннеры</th><th class="n">Ошибки баннеров</th><th class="n">Ролики</th><th class="n">Ошибки роликов</th></tr></thead>
<tbody>${rows || '<tr><td colspan="10">Пока нет данных</td></tr>'}</tbody></table></div>
<p class="muted">Счётчики — за всё время. Код телефона — первые 6 символов зашифрованного ID, настоящий ID не хранится.</p>
</main></body></html>`;
  return new Response(html, { headers: { "Content-Type": "text/html; charset=utf-8", "Cache-Control": "no-store" } });
}
